{{- /* Alias for code-file shortcode */ -}} {{- $source := .Get "source" | default (.Get "file") | default (.Get 0) -}} {{- $lang := .Get "lang" | default (.Get "language") -}} {{- $options := .Get "options" | default "" -}} {{- $title := .Get "title" -}} {{- if not $source -}} {{- errorf "Shortcode 'code' requires a 'source' parameter pointing to a file path. Position: %s" .Position -}} {{- end -}} {{- if not $lang -}} {{- $ext := path.Ext $source | strings.TrimPrefix "." -}} {{- if eq $ext "py" }}{{ $lang = "python" }} {{- else if eq $ext "js" }}{{ $lang = "javascript" }} {{- else if eq $ext "ts" }}{{ $lang = "typescript" }} {{- else if eq $ext "rs" }}{{ $lang = "rust" }} {{- else if eq $ext "cpp" }}{{ $lang = "cpp" }} {{- else if eq $ext "c" }}{{ $lang = "c" }} {{- else if eq $ext "sh" }}{{ $lang = "bash" }} {{- else if eq $ext "yml" }}{{ $lang = "yaml" }} {{- else if eq $ext "toml" }}{{ $lang = "toml" }} {{- else if eq $ext "json" }}{{ $lang = "json" }} {{- else if $ext }}{{ $lang = $ext }} {{- else }}{{ $lang = "plaintext" }}{{ end -}} {{- end -}} {{- $content := "" -}} {{- with .Page.Resources.Get $source -}} {{- $content = .Content -}} {{- else -}} {{- with resources.Get $source -}} {{- $content = .Content -}} {{- else -}} {{- if fileExists $source -}} {{- $content = readFile $source -}} {{- end -}} {{- end -}} {{- end -}} {{- if $content -}} {{- $codeSnippet := trim $content "\n\r" -}} {{- $displayTitle := $title | default (path.Base $source) -}}
📄 {{ $displayTitle }}
{{ $lang }}
{{ highlight $codeSnippet $lang $options }}
{{- else -}}
⚠️ Code File Missing: Unable to locate {{ $source }}. Ensure the file path is relative to the page bundle, assets directory, or site root.
{{- end -}}
